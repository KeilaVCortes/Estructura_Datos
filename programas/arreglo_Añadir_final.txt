import java.util.Scanner;

public class ArregloEjemploFinal {
    public static void main(String[] args) {
        // Creamos un escáner para recibir entrada del usuario
        Scanner scanner = new Scanner(System.in);
        
        // Creamos un arreglo con un tamaño de 10 para almacenar los números
        int[] numeros = new int[10];
        
        // Solicitamos 10 números y los agregamos al final del arreglo
        System.out.println("Por favor, ingrese 10 números:");
        for (int i = 0; i < 10; i++) {
            System.out.print("Número " + (i + 1) + ": ");
            numeros[i] = scanner.nextInt(); // Guardamos cada número en el arreglo en orden
        }
        
        // Mostramos los números ingresados (en el orden en que se introdujeron)
        System.out.println("\nLos números ingresados son:");
        for (int i = 0; i < 10; i++) {
            System.out.println("Número " + (i + 1) + ": " + numeros[i]);
        }
        
        // Cerramos el escáner
        scanner.close();
    }
}
