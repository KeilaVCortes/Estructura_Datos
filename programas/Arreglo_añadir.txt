//Pidiendo 10 números a un usuario


import java.util.Scanner;

public class VectorEjemplo {
    public static void main(String[] args) {
        // Creamos un escáner para recibir entrada del usuario
        Scanner scanner = new Scanner(System.in);
        
        // Creamos un arreglo para almacenar los 10 números
        int[] numeros = new int[10];
        
        // Solicitamos los 10 números
        System.out.println("Por favor, ingrese 10 números:");
        for (int i = 0; i < numeros.length; i++) {
            System.out.print("Número " + (i + 1) + ": ");
            numeros[i] = scanner.nextInt(); // Guardamos cada número en el vector
        }
        
        // Mostramos los números ingresados
        System.out.println("\nLos números ingresados son:");
        for (int i = 0; i < numeros.length; i++) {
            System.out.println("Número " + (i + 1) + ": " + numeros[i]);
        }
        
        // Cerramos el escáner
        scanner.close();
    }
}
