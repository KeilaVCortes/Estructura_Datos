//pedir el numero de usuario del jugador y eliminarlo junto con su puntaje


import java.util.Scanner;
import java.util.Arrays;

public class JugadoresPuntajes {
    public static void main(String[] args) {
        // Arreglo de puntajes
        int[] puntajes = {0, 70, 50, 10, 80, 20, 60, 30, 40, 90};
        // Arreglo de jugadores
        String[] jugadores = {"J1", "J2", "J3", "J4", "J5", "J6", "J7", "J8", "J9", "J10"};

        // Crear un objeto Scanner para leer la entrada del usuario
        Scanner scanner = new Scanner(System.in);

        // Pedir el número de usuario del jugador
        System.out.print("Ingrese el número de jugador (1-10) que desea eliminar: ");
        int numeroJugador = scanner.nextInt();

        // Verificar que el número esté dentro del rango válido
        if (numeroJugador >= 1 && numeroJugador <= 10) {
            // Convertir el número de jugador a índice (restar 1)
            int posicion = numeroJugador - 1;

            // Eliminar el jugador y su puntaje
            jugadores = eliminarElemento(jugadores, posicion);
            puntajes = eliminarElemento(puntajes, posicion);

            // Imprimir los arreglos actualizados
            System.out.println("Jugador eliminado. Listado actualizado:");
            System.out.println("Jugadores: " + Arrays.toString(jugadores));
            System.out.println("Puntajes: " + Arrays.toString(puntajes));
        } else {
            // Mensaje de error si el número no está en el rango válido
            System.out.println("Número de jugador inválido. Debe ser entre 1 y 10.");
        }

        // Cerrar el escáner
        scanner.close();
    }

    // Método para eliminar un elemento de un arreglo de String
    public static String[] eliminarElemento(String[] arreglo, int indice) {
        if (arreglo == null || indice < 0 || indice >= arreglo.length) {
            return arreglo;
        }

        String[] nuevoArreglo = new String[arreglo.length - 1];

        for (int i = 0, j = 0; i < arreglo.length; i++) {
            if (i != indice) {
                nuevoArreglo[j++] = arreglo[i];
            }
        }
        return nuevoArreglo;
    }

    // Método para eliminar un elemento de un arreglo de int
    public static int[] eliminarElemento(int[] arreglo, int indice) {
        if (arreglo == null || indice < 0 || indice >= arreglo.length) {
            return arreglo;
        }

        int[] nuevoArreglo = new int[arreglo.length - 1];

        for (int i = 0, j = 0; i < arreglo.length; i++) {
            if (i != indice) {
                nuevoArreglo[j++] = arreglo[i];
            }
        }
        return nuevoArreglo;
    }
}
