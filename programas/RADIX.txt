import java.util.Scanner;

public class AlgoritmosOrdenacion {
    public static void main(String[] args) {
        // Creación de un arreglo
        int[] elementos = { 38, 18, 50, 8, 6, 100, 94, 90, 22, 10 };

        // Mostrar elementos
        System.out.println("Vector Inicial:");
        for (int i = 0; i < elementos.length; i++) {
            System.out.println("Elemento " + i + ": " + elementos[i]);
        }

        // Declaración de variable OPCION
        Scanner teclado = new Scanner(System.in);
        int opcion; // Declarando la variable para almacenar la opción

        do {
            System.out.println("          +*+*+* MENU PRINCIPAL *+*+*");
            System.out.println("   A) ALGORITMOS DE ORDENACION INTERNA");
            System.out.println("      1.- Burbuja (BubbleSort)");
            System.out.println("      2.- QuickSort");
            System.out.println("      3.- ShellSort");
            System.out.println("      4.- RadixSort");
            System.out.println("   B) ALGORITMOS DE ORDENACION EXTERNA");
            System.out.println("      5.- Intercalación");
            System.out.println("      6.- Mezcla directa (MergeSort)");
            System.out.println("      7.- Mezcla Natural");
            System.out.println("   8.- Salida");
            System.out.println("Elige una opción:");

            // Leer opción
            opcion = teclado.nextInt();

            // Llamado a las funciones
            switch (opcion) {
                case 1: // BURBUJA
                    burbuja(elementos);
                    System.out.println("Vector ordenado con Burbuja:");
                    mostrarVector(elementos);
                    break;
                case 2: // QUICKSORT
                    quickSort(elementos, 0, elementos.length - 1);
                    System.out.println("Vector ordenado con QuickSort:");
                    mostrarVector(elementos);
                    break;
                case 3: // SHELLSORT
                    shellSort(elementos);
                    System.out.println("Vector ordenado con ShellSort:");
                    mostrarVector(elementos);
                    break;
                case 4: // RADIXSORT
                    radixSort(elementos);
                    System.out.println("Vector ordenado con RadixSort:");
                    mostrarVector(elementos);
                    break;
                case 5: // INTERCALACIÓN (no implementado)
                    System.out.println("Vector ordenado con Intercalación (pendiente de implementación):");
                    break;
                case 6: // MEZCLA DIRECTA (no implementado)
                    System.out.println("Vector ordenado con Mezcla Directa (pendiente de implementación):");
                    break;
                case 7: // MEZCLA NATURAL (no implementado)
                    System.out.println("Vector ordenado con Mezcla Natural (pendiente de implementación):");
                    break;
                case 8: // SALIDA
                    System.out.println("Fin del programa");
                    break;
                default: // OPCIÓN NO VÁLIDA
                    System.out.println("Opción no válida. Por favor, selecciona una opción entre 1 y 8.");
                    break;
            }
        } while (opcion != 8); // Condición para salir

        teclado.close(); // Cerrar el Scanner
    }

    // Método para mostrar el vector
    public static void mostrarVector(int[] vector) {
        for (int i = 0; i < vector.length; i++) {
            System.out.println("Elemento " + i + ": " + vector[i]);
        }
    }

    // Algoritmo Burbuja
    public static void burbuja(int[] vector) {
        int n = vector.length;
        for (int i = 0; i < n - 1; i++) {
            for (int j = 0; j < n - 1 - i; j++) {
                if (vector[j] > vector[j + 1]) {
                    // Intercambio
                    int temp = vector[j];
                    vector[j] = vector[j + 1];
                    vector[j + 1] = temp;
                }
            }
        }
    }

    // Algoritmo QuickSort
    public static void quickSort(int[] vector, int inicio, int fin) {
        if (inicio < fin) {
            int indicePivote = particion(vector, inicio, fin);
            quickSort(vector, inicio, indicePivote - 1);
            quickSort(vector, indicePivote + 1, fin);
        }
    }

    // Método para la partición de QuickSort
    public static int particion(int[] vector, int inicio, int fin) {
        int pivote = vector[fin];
        int i = inicio - 1;

        for (int j = inicio; j < fin; j++) {
            if (vector[j] <= pivote) {
                i++;
                // Intercambio
                int temp = vector[i];
                vector[i] = vector[j];
                vector[j] = temp;
            }
        }

        // Intercambiar el pivote con el elemento siguiente al último menor
        int temp = vector[i + 1];
        vector[i + 1] = vector[fin];
        vector[fin] = temp;

        return i + 1;
    }

    // Algoritmo ShellSort
    public static void shellSort(int[] vector) {
        int n = vector.length;
        for (int gap = n / 2; gap > 0; gap /= 2) {
            for (int i = gap; i < n; i++) {
                int temp = vector[i];
                int j;
                for (j = i; j >= gap && vector[j - gap] > temp; j -= gap) {
                    vector[j] = vector[j - gap];
                }
                vector[j] = temp;
            }
        }
    }

    // Algoritmo RadixSort
    public static void radixSort(int[] vector) {
        int max = obtenerMaximo(vector);
        for (int exp = 1; max / exp > 0; exp *= 10) {
            countingSortPorDigito(vector, exp);
        }
    }

    // Método auxiliar para obtener el máximo valor en el arreglo
    public static int obtenerMaximo(int[] vector) {
        int max = vector[0];
        for (int i = 1; i < vector.length; i++) {
            if (vector[i] > max) {
                max = vector[i];
            }
        }
        return max;
    }

    // Counting Sort usado por RadixSort
    public static void countingSortPorDigito(int[] vector, int exp) {
        int n = vector.length;
        int[] salida = new int[n];
        int[] conteo = new int[10];

        // Contar las ocurrencias
        for (int i = 0; i < n; i++) {
            conteo[(vector[i] / exp) % 10]++;
        }

        // Transformar conteo a posiciones reales
        for (int i = 1; i < 10; i++) {
            conteo[i] += conteo[i - 1];
        }

        // Construir el arreglo de salida
        for (int i = n - 1; i >= 0; i--) {
            salida[conteo[(vector[i] / exp) % 10] - 1] = vector[i];
            conteo[(vector[i] / exp) % 10]--;
        }

        // Copiar el arreglo de salida al original
        for (int i = 0; i < n; i++) {
            vector[i] = salida[i];
        }
    }
}
