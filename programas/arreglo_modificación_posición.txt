//código en un arreglo simple en donde se modifique el valor de un elemento dependiendo del usuario


import java.util.Scanner;

public class ModificarArreglo {
    public static void main(String[] args) {
        // Creamos un escáner para recibir entrada del usuario
        Scanner scanner = new Scanner(System.in);
        
        // Creamos un arreglo con un tamaño de 10 para almacenar los números
        int[] numeros = new int[10];
        
        // Inicializamos el arreglo con valores predeterminados (por ejemplo, del 1 al 10)
        for (int i = 0; i < numeros.length; i++) {
            numeros[i] = i + 1;
        }
        
        // Mostramos el arreglo inicial
        System.out.println("El arreglo inicial es:");
        for (int i = 0; i < numeros.length; i++) {
            System.out.println("Índice " + i + ": " + numeros[i]);
        }

        // Pedimos al usuario que ingrese el índice y el nuevo valor
        System.out.print("\nIngrese el índice (0-9) del valor que desea modificar: ");
        int indice = scanner.nextInt();
        
        // Verificamos que el índice esté dentro del rango
        if (indice >= 0 && indice < numeros.length) {
            System.out.print("Ingrese el nuevo valor para el índice " + indice + ": ");
            int nuevoValor = scanner.nextInt();
            numeros[indice] = nuevoValor; // Modificamos el valor en el índice especificado
        } else {
            System.out.println("Índice fuera de rango.");
        }
        
        // Mostramos el arreglo después de la modificación
        System.out.println("\nEl arreglo después de la modificación es:");
        for (int i = 0; i < numeros.length; i++) {
            System.out.println("Índice " + i + ": " + numeros[i]);
        }
        
        // Cerramos el escáner
        scanner.close();
    }
}
